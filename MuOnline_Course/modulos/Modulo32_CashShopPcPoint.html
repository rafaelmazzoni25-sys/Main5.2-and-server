void CCashShop::LoadPackage(char* path) {
  CMemScript* lpMemScript = new CMemScript;
  if(lpMemScript->SetBuffer(path) == 0) { ErrorMessageBox(...); delete lpMemScript; return; }
  this->m_CashShopPackageCount = 0;
  while(true) {
    if(lpMemScript->GetToken() == TOKEN_END) break;
    if(strcmp("end",lpMemScript->GetString()) == 0) break;
    CASH_SHOP_PACKAGE_INFO info{};
    info.Category = lpMemScript->GetNumber();
    info.BaseIndex = lpMemScript->GetAsNumber();
    info.MainIndex = lpMemScript->GetAsNumber();
    info.ItemIndex = lpMemScript->GetAsNumber();
    info.CoinIndex = lpMemScript->GetAsNumber();
    info.CoinValue = lpMemScript->GetAsNumber();
    info.BonusGP = lpMemScript->GetAsNumber();
    for(int n=0;n < MAX_CASH_SHOP_PACKAGE_PRODUCT;n++) info.ProductBaseIndex[n] = lpMemScript->GetAsNumber();
    for(int n=0;n < MAX_CASH_SHOP_PACKAGE_PRODUCT;n++) info.ProductMainIndex[n] = lpMemScript->GetAsNumber();
    this->SetPackageInfo(info);
  }
}

CCashShop::MainProc – Expiração de itens periódicos
📘 O Que é?

Loop que percorre inventário e remove itens com m_PeriodicItemTime expirado.

🎯 Para Que Serve?

Garante que buffs/itens temporários do Cash Shop sejam removidos quando o tempo acaba.

💻 Como Funciona?

for(int n=OBJECT_START_USER;n < MAX_OBJECT;n++) {
  if(gObjIsConnectedGP(n) == 0) continue;
  LPOBJ lpObj = &gObj[n];
  for(int i=0;i < INVENTORY_SIZE;i++) {
    if(lpObj->Inventory[i].IsItem() == 0) continue;
    if(lpObj->Inventory[i].m_IsPeriodicItem == 0) continue;
    if(lpObj->Inventory[i].m_LoadPeriodicItem == 0) continue;
    if(lpObj->Inventory[i].m_PeriodicItemTime <= ((int)time(0))) {
      gItemManager.InventoryDelItem(lpObj->Index,i);
      gItemManager.GCItemDeleteSend(lpObj->Index,i,1);
      gItemManager.UpdateInventoryViewport(lpObj->Index,i);
    }
  }
}

void CPcPoint::CGPcPointItemBuyRecv(PMSG_PC_POINT_ITEM_BUY_RECV* lpMsg,int aIndex) {
  if(gServerInfo.m_PcPointSwitch == 0) return;
  LPOBJ lpObj = &gObj[aIndex];
  if(((lpObj->PcPointTransaction[1]==0)?(lpObj->PcPointTransaction[1]++):lpObj->PcPointTransaction[1]) != 0) return;
  SDHP_PC_POINT_ITEM_BUY_SEND pMsg; pMsg.header.set(0x19,0x01,sizeof(pMsg));
  pMsg.index = aIndex; memcpy(pMsg.account,lpObj->Account,sizeof(pMsg.account));
  pMsg.slot = lpMsg->slot;
  gDataServerConnection.DataSend((BYTE*)&pMsg,pMsg.header.size);
}

Implementação em Unreal Engine 5

  1. Modelar UCashCatalogAsset (DataAsset) contendo arrays de pacotes/produtos carregados de JSON (equivalente aos scripts).
  2. Criar UCashShopComponent (ActorComponent) no APlayerController para abrir UI, listar produtos e disparar ServerBuyCashItem com validações de interface.
  3. Implementar backend Node.js/SQL com rotas /cash/catalog e /cash/buy; o cliente UE5 chama via VaRest e recebe autorização antes de spawnar o item.
  4. Adicionar verificação de itens temporários em um FTimerManager no AGameState, removendo itens expirados no inventário do ACharacter (replicação usando OnRep_Inventory).
  5. Para PcPoint, criar UPcPointService que consulta /pcpoint/catalog e /pcpoint/buy; replicar saldo em APlayerState e usar Server/Client RPCs para atualizar UI.

Passo a Passo em Unreal Engine 5

  1. Importar catálogos JSON (pacotes/produtos) como DataAssets e expor em Blueprints para designers.
  2. Construir UI UMG de Cash Shop com filtros de categoria e botão "Comprar" que chama ServerBuyCashItem; bloquear quando InterfaceState != None, espelhando as checagens originais.
